cmake_minimum_required(VERSION 3.24)
project(JSTILocalLLMRuntime LANGUAGES C CXX)

if(NOT JSTI_LLAMA_SOURCE OR NOT EXISTS "${JSTI_LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Supply the authenticated b6500 source with -DJSTI_LLAMA_SOURCE=...")
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This native desktop runtime packager supports Windows and Linux only")
endif()
if(WIN32 AND NOT MSVC)
    message(FATAL_ERROR "Use the existing MSVC-compatible native Windows toolchain (MSVC or clang-cl)")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    # The confined DLL loader searches only its own directory and System32.
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_CPU_HBM OFF CACHE BOOL "" FORCE)
foreach(backend CUDA HIP VULKAN METAL OPENCL SYCL RPC BLAS MUSA CANN WEBGPU ZDNN ACCELERATE)
    set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
# Baseline portable CPU instructions; no build-machine AVX assumptions.
foreach(instruction SSE42 AVX AVX_VNNI AVX2 AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16
        FMA F16C BMI2 AMX_TILE AMX_INT8 AMX_BF16)
    set(GGML_${instruction} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${JSTI_LLAMA_SOURCE}" llama-build EXCLUDE_FROM_ALL)
set_target_properties(llama PROPERTIES CXX_VISIBILITY_PRESET default VISIBILITY_INLINES_HIDDEN OFF)
add_library(jsti_llama_runtime SHARED runtime-anchor.c)
target_link_libraries(jsti_llama_runtime PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,llama>")
set_target_properties(jsti_llama_runtime PROPERTIES OUTPUT_NAME llama)
if(WIN32)
    target_sources(jsti_llama_runtime PRIVATE exports.def)
else()
    set_target_properties(jsti_llama_runtime PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS jsti_llama_runtime RUNTIME DESTINATION . LIBRARY DESTINATION .)
